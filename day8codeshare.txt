import time

    schema(empSchema). \

csvstream_withcheckpoint.py

from pyspark.sql import SparkSession
from pyspark.sql.types import *
from pyspark.sql.functions import input_file_name,regexp_extract
import time
spark=SparkSession.builder.appName("structured-streaming-csv-app-1").getOrCreate()
empSchema=StructType([StructField("empid",IntegerType()),StructField("name",StringType()),StructField("designation",StringType())])
stream_df=spark.readStream.format("csv"). \
    schema(empSchema). \
    load("/home/student/structureddataset/csvstreamingdata/")
stream_df_with_filename=stream_df.withColumn("input_file",regexp_extract(input_file_name(),r"([^/]+)$",1))
streaming_query=stream_df_with_filename.writeStream.  \
    option("truncate",False).\
    option("checkpointLocation","/home/student/csv-checkpoint-1"). \
    format("console")\
 .trigger(processingTime="20 seconds").start()
print("streaming query is started")
time.sleep(5*60)
streaming_query.stop()
print("streaming query stopped")
