import pandas as pd

emp_df=pd.read_csv("/home/student/pythontraining/dataset/employee_1.csv")
#print(df.to_string(index=False))
dept_df=pd.read_json("/home/student/pythontraining/dataset/dept_1.json")
merge_df=pd.merge(emp_df,dept_df,left_on="dept_id",right_on="department_id",how="inner")
print(merge_df)
merge_df_1=merge_df[["emp_id","name","dept_name","salary"]]
print(merge_df_1)
result=pd.pivot_table(merge_df_1,index="dept_name",values="salary",aggfunc='sum')
print(result)
