from pyspark.sql.functions import *
from pyspark.sql import SparkSession
spark=SparkSession.builder.appName("structured-streaming-csv-app-1")\
    .config("spark.jars.packages","org.apache.spark:spark-sql-kafka-0-10_2.13:4.2.0") \
    .getOrCreate()
large_df=(
    spark.range(0,100000)
    .withColumn("department",
                when(col("id")%4==0,"IT")
                .when(col("id")%4==1,"HR")
                .when(col("id")%4==2,"Finance")
                .otherwise("Sales")
    )
    .withColumn("salary",(col("id")%50)+20000)

)
print("Records:",large_df.count())
print("Partitions: ",large_df.rdd.getNumPartitions())
large_df.show(10)
