from pyspark.sql import SparkSession
from pyspark.sql.types import *
from pyspark.sql.functions import col,from_json
import time
spark=SparkSession.builder.appName("structured-streaming-csv-app-1")\
    .config("spark.jars.packages","org.apache.spark:spark-sql-kafka-0-10_2.13:4.2.0") \
    .getOrCreate()
empSchema=StructType([StructField("empid",IntegerType()),\
                      StructField("name",StringType()) \
                      ,StructField("designation",StringType()) \
                      ,StructField("salary",DoubleType())])
kafka_stream_df=spark.readStream.format("kafka"). \
    option("kafka.bootstrap.servers","localhost:9092") . \
    option("subscribe","emp-data-topic"). \
    load()
json_df=kafka_stream_df.select(col("value").cast("string").alias("json_data"))
parsed_df= json_df.select(from_json(col("json_data"),empSchema).alias("emp")).select("emp.*")
transfomed_df=parsed_df.select("empid","name","salary").where("salary>40000")

streaming_query=transfomed_df.writeStream.  \
    option("truncate",False).\
    option("checkpointLocation","/home/student/csv-checkpoint-3"). \
    format("csv") \
    .option("path","/home/student/dataframe-out/etl-csv-out-1") \
    .option("header",True) \
    .trigger(processingTime="20 seconds").start()
print("streaming query is started")
time.sleep(5*60)
streaming_query.stop()
print("streaming query stopped")
