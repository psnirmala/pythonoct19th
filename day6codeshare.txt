from pyspark import SparkContext
import time
sc=SparkContext(appName="first-app")
sc.setLogLevel("WARN")
rdd1=sc.textFile("/home/student/pythontraining/textdata/words.txt")
word_count_rdd=rdd1.flatMap(lambda line:line.split(" ")).map(lambda word:(word,1)).reduceByKey(lambda x,y:x+y)
for key,value in word_count_rdd.collect():
    print(key,value)
time.sleep(5*60)
print("application completed")
