functional_prog_1.py


def add(x,y):
    return x+y

def subtract(x,y):
    return x-y

def compute(x,y,f):
    return f(x,y)

sum=compute(10,5,add)
diff=compute(20,12,subtract)
print(f"Sum={sum} Difference={diff}")
product=compute(5,4,lambda a,b:a*b)
print(f"Product={product}")
quotient=compute(20,4,lambda a,b:a/b)
print(f"Quotient={quotient}")



maximum=compute(10,8,lambda a,b:a if a>b else b)
print(f"Maximum :{maximum}")

def get_function(op):
    if op=="add":
        return lambda a,b:a+b
    elif op=="subtract":
        return lambda a,b:a-b
    else:
        return lambda  a,b:None

operation=get_function("add")
print(operation(10,2))
operation=get_function("subtract")
print(operation(10,2))

operation=get_function("mulitply")
print(operation(10,2))
