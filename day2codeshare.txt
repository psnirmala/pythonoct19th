class Employee(Person):
    def __init__(self,name,age,gender,address,id):
        super().__init__(name,age,gender,address)
        self.id=id

    def display(self):
        super().display()
        print(f"Employee id {self.id}")

address1=Address("Hebbal","Bangalore")
employee1=Employee("Arvind",27,"Male",address1,32225)
employee1.display()
